Flows for hash sanity checks:
    flow 1. HSET(k, f1, v1, f2, v2) -> HGET(k, f1, f2) -> return v1 v2
    flow 2. HSET(k, f1, v1, f2, v2, f3, v3) -> HGETALL(k) -> return all data
    flow 3. HSET(k, f1, v1, f2, v2) -> HKEYS(k) -> return f1 f2

    flow 4. HSET(k, f1, v1) -> HDEL(k, f1, f2) -> delete field f1, not f2

    flow 5. HSET(k, f1, v1) -> HEXISTS(k, f1, f2) -> field f1 exists, f2 does not

    flow 6. HSET(k, f1, v1) -> HLEN(k) -> length 1

    flow 7. HSET(k, f1, v1, f2, v2) -> HSTRLEN(k, f1, f2) -> length 2 2
    flow 8. HSET(k, f1, v1, f2, v2) -> HSTRLEN(k, f3) -> field does not exist

    flow 9. HGET(k, f1) -> key not found
    flow 10. HGETALL(k) -> key not found
    flow 11. HKEYS(k) -> key not found
    flow 12. HDEL(k, f1) -> key not found
    flow 13. HEXISTS(k, f1) -> key not found
    flow 14. HLEN(k) -> key not found
    flow 15. HSTRLEN(k, f1) -> key not found
